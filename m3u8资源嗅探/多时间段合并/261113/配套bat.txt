@echo off
chcp 65001
setlocal enabledelayedexpansion
cd /d "%~dp0"

set "TXT_DIR=C:\Users\sjopfops\Downloads"

echo ==============================================
echo 程序目录：%cd%
echo 扫描目录：%TXT_DIR%
echo ==============================================

if not exist "%TXT_DIR%" (
    echo ❌ 目录不存在
    exit /b
)

for /f "delims=" %%f in ('dir /b /on "%TXT_DIR%\*.txt" 2^>nul') do (
    set "TXT_FULL=!TXT_DIR!\%%f"
    echo.
    echo ==============================================
    echo 正在处理：%%f
    echo 文件路径：!TXT_FULL!
    echo ==============================================

    for /f "usebackq delims=" %%L in ("!TXT_FULL!") do (
        if not "%%L"=="" (
            echo 执行命令：[%%L]
            %%L
            echo --------------------------------
        )
    )

    echo ✅ %%f 执行完毕，准备删除
    del "!TXT_FULL!"
    echo 🗑️ 已删除 %%f
    echo.
    choice /c yn /t 10 /d y /m "是否处理下一个txt？[Y=继续 / N=结束 / 10秒超时默认继续]"
    if errorlevel 2 goto :end_all
)
:end_all

echo ==============================================
echo ✅ 所有txt任务处理并删除完成
echo ==============================================
